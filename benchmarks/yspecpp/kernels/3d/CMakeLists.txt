cmake_minimum_required(VERSION 3.24)
project(PhaseDKernelBenchmark LANGUAGES CXX)
set(SOLVER "" CACHE STRING "dspec or yspecpp")
set(DSPEC_SOURCE_DIR "" CACHE PATH "DSpecM1D checkout")
set(YSPEC_SOURCE_DIR "" CACHE PATH "YSpecpp checkout")
set(DEPENDENCY_ROOT "" CACHE PATH "Verified pinned dependency source cache")
set(DSPEC_PLANETMODEL_SOURCE_DIR "" CACHE PATH "Optional DSpec-pinned planetmodel checkout")
set(YSPEC_PLANETMODEL_SOURCE_DIR "" CACHE PATH "Optional YSpec-pinned planetmodel checkout")
foreach(dep numericconcepts interpolation fftwpp gaussquad gshtrans)
  string(TOUPPER "${dep}" upper)
  if(NOT DEPENDENCY_ROOT STREQUAL "" AND IS_DIRECTORY "${DEPENDENCY_ROOT}/${dep}-src")
    set(FETCHCONTENT_SOURCE_DIR_${upper} "${DEPENDENCY_ROOT}/${dep}-src" CACHE PATH "" FORCE)
  endif()
endforeach()
if(NOT DSPEC_PLANETMODEL_SOURCE_DIR STREQUAL "" AND IS_DIRECTORY "${DSPEC_PLANETMODEL_SOURCE_DIR}")
  set(FETCHCONTENT_SOURCE_DIR_PLANETMODEL "${DSPEC_PLANETMODEL_SOURCE_DIR}" CACHE PATH "" FORCE)
endif()
if(SOLVER STREQUAL "dspec")
  set(DSPECM1D_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
  set(DSPECM1D_BUILD_TUTORIALS OFF CACHE BOOL "" FORCE)
  set(DSPECM1D_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(DSPECM1D_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(DSPECM1D_BUILD_LOVE_NUMBERS OFF CACHE BOOL "" FORCE)
  add_subdirectory("${DSPEC_SOURCE_DIR}" dspec)
  add_executable(phase_d_adapter dspec_3d_adapter.cpp)
  target_link_libraries(phase_d_adapter PRIVATE DSpecM1D)
elseif(SOLVER STREQUAL "yspecpp")
  set(YSPEC_NUMERICCONCEPTS_TAG e03a2889d43a03660d0879f46dda1406d90c9517 CACHE STRING "" FORCE)
  set(YSPEC_INTERPOLATION_TAG d5a1711ca6a89fc8ad8148fe13212398fe569bc0 CACHE STRING "" FORCE)
  set(YSPEC_FFTWPP_TAG cb152c0e85f3822493c2d9e85a56307ac7b4f1c0 CACHE STRING "" FORCE)
  set(YSPEC_GAUSSQUAD_TAG 7d4786d192f68077cd99a34dcdf1379ed1eeac9a CACHE STRING "" FORCE)
  set(YSPEC_GSHTRANS_TAG 8dd116df8df4222fe9b10a40c7213d34f2a274fb CACHE STRING "" FORCE)
  set(YSPEC_PLANETMODEL_TAG fa6e230fde25152d40f27d9839de44c4f3182a24 CACHE STRING "" FORCE)
  if(NOT YSPEC_PLANETMODEL_SOURCE_DIR STREQUAL "" AND IS_DIRECTORY "${YSPEC_PLANETMODEL_SOURCE_DIR}")
    set(FETCHCONTENT_SOURCE_DIR_PLANETMODEL "${YSPEC_PLANETMODEL_SOURCE_DIR}" CACHE PATH "" FORCE)
  endif()
  set(PLANETMODEL_WITH_NETCDF OFF CACHE BOOL "" FORCE)
  set(GSHTRANS_WITH_BLAS OFF CACHE STRING "" FORCE)
  set(YSPEC_WITH_OPENMP ON CACHE BOOL "" FORCE)
  set(YSPEC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory("${YSPEC_SOURCE_DIR}" yspec)
  add_executable(phase_d_adapter yspec_3d_adapter.cpp)
  add_executable(yspec_kernel_timing yspec_kernel_timing.cpp)
  target_include_directories(phase_d_adapter PRIVATE "${YSPEC_SOURCE_DIR}/src")
  target_link_libraries(phase_d_adapter PRIVATE YSpec::YSpec YSpecInternal)
  target_include_directories(yspec_kernel_timing PRIVATE "${YSPEC_SOURCE_DIR}/src")
  target_link_libraries(yspec_kernel_timing PRIVATE YSpec::YSpec YSpecInternal)
else()
  message(FATAL_ERROR "SOLVER must be dspec or yspecpp")
endif()
