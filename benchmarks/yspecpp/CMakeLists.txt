cmake_minimum_required(VERSION 3.24)
project(DSpecM1D_YSpecppBenchmark LANGUAGES CXX)

set(BENCHMARK_SOLVER "" CACHE STRING "Build adapter for dspec or yspecpp")
set_property(CACHE BENCHMARK_SOLVER PROPERTY STRINGS dspec yspecpp)
set(DSPEC_SOURCE_DIR "" CACHE PATH "DSpecM1D source checkout")
set(YSPEC_SOURCE_DIR "" CACHE PATH "Immutable YSpecpp source snapshot")
set(VERIFIED_DEPENDENCY_ROOT "" CACHE PATH "Verified dependency source cache")

if(NOT BENCHMARK_SOLVER STREQUAL "dspec" AND
   NOT BENCHMARK_SOLVER STREQUAL "yspecpp")
  message(FATAL_ERROR "Set BENCHMARK_SOLVER to dspec or yspecpp")
endif()
if(NOT IS_DIRECTORY "${VERIFIED_DEPENDENCY_ROOT}")
  message(FATAL_ERROR "Set VERIFIED_DEPENDENCY_ROOT to a SHA-verified source cache")
endif()

if(BENCHMARK_SOLVER STREQUAL "dspec")
  if(NOT IS_DIRECTORY "${DSPEC_SOURCE_DIR}/DSpecM1D")
    message(FATAL_ERROR "Set DSPEC_SOURCE_DIR to the DSpecM1D checkout")
  endif()
  set(FETCHCONTENT_SOURCE_DIR_NUMERICCONCEPTS "${VERIFIED_DEPENDENCY_ROOT}/numericconcepts-src" CACHE PATH "" FORCE)
  set(FETCHCONTENT_SOURCE_DIR_INTERPOLATION "${VERIFIED_DEPENDENCY_ROOT}/interpolation-src" CACHE PATH "" FORCE)
  set(FETCHCONTENT_SOURCE_DIR_FFTWPP "${VERIFIED_DEPENDENCY_ROOT}/fftwpp-src" CACHE PATH "" FORCE)
  set(FETCHCONTENT_SOURCE_DIR_GAUSSQUAD "${VERIFIED_DEPENDENCY_ROOT}/gaussquad-src" CACHE PATH "" FORCE)
  set(FETCHCONTENT_SOURCE_DIR_GSHTRANS "${VERIFIED_DEPENDENCY_ROOT}/gshtrans-src" CACHE PATH "" FORCE)
  set(DSPECM1D_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
  set(DSPECM1D_BUILD_TUTORIALS OFF CACHE BOOL "" FORCE)
  set(DSPECM1D_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(DSPECM1D_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(DSPECM1D_BUILD_LOVE_NUMBERS OFF CACHE BOOL "" FORCE)
  add_subdirectory("${DSPEC_SOURCE_DIR}" dspecm1d EXCLUDE_FROM_ALL)
  add_executable(dspec_adapter dspec_adapter.cpp)
  target_link_libraries(dspec_adapter PRIVATE DSpecM1D)
else()
  if(NOT IS_DIRECTORY "${YSPEC_SOURCE_DIR}/YSpec")
    message(FATAL_ERROR "Set YSPEC_SOURCE_DIR to the immutable YSpecpp source snapshot")
  endif()
  set(FETCHCONTENT_SOURCE_DIR_NUMERICCONCEPTS "${VERIFIED_DEPENDENCY_ROOT}/numericconcepts-src" CACHE PATH "" FORCE)
  set(FETCHCONTENT_SOURCE_DIR_INTERPOLATION "${VERIFIED_DEPENDENCY_ROOT}/interpolation-src" CACHE PATH "" FORCE)
  set(FETCHCONTENT_SOURCE_DIR_FFTWPP "${VERIFIED_DEPENDENCY_ROOT}/fftwpp-src" CACHE PATH "" FORCE)
  set(FETCHCONTENT_SOURCE_DIR_GAUSSQUAD "${VERIFIED_DEPENDENCY_ROOT}/gaussquad-src" CACHE PATH "" FORCE)
  set(FETCHCONTENT_SOURCE_DIR_GSHTRANS "${VERIFIED_DEPENDENCY_ROOT}/gshtrans-src" CACHE PATH "" FORCE)
  set(YSPEC_NUMERICCONCEPTS_TAG e03a2889d43a03660d0879f46dda1406d90c9517 CACHE STRING "" FORCE)
  set(YSPEC_INTERPOLATION_TAG d5a1711ca6a89fc8ad8148fe13212398fe569bc0 CACHE STRING "" FORCE)
  set(YSPEC_FFTWPP_TAG cb152c0e85f3822493c2d9e85a56307ac7b4f1c0 CACHE STRING "" FORCE)
  set(YSPEC_GAUSSQUAD_TAG 7d4786d192f68077cd99a34dcdf1379ed1eeac9a CACHE STRING "" FORCE)
  set(YSPEC_GSHTRANS_TAG 8dd116df8df4222fe9b10a40c7213d34f2a274fb CACHE STRING "" FORCE)
  set(YSPEC_PLANETMODEL_TAG fa6e230fde25152d40f27d9839de44c4f3182a24 CACHE STRING "" FORCE)
  set(PLANETMODEL_WITH_NETCDF OFF CACHE BOOL "" FORCE)
  set(GSHTRANS_WITH_BLAS OFF CACHE STRING "" FORCE)
  set(YSPEC_WITH_OPENMP ON CACHE BOOL "" FORCE)
  set(YSPEC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory("${YSPEC_SOURCE_DIR}" yspecpp EXCLUDE_FROM_ALL)
  add_executable(yspecpp_adapter yspecpp_adapter.cpp)
  target_link_libraries(yspecpp_adapter PRIVATE YSpec::YSpec)
endif()
